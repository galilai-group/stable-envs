import hashlib
import json
from pathlib import Path

root = Path.cwd()
e = root / 'docs/research/semantic_recovery_evidence/craftax_gpu_6679082'
verified = json.loads((e / 'verified-gpu.json').read_text())
assert verified['slurm_state'] == 'COMPLETED' and verified['exit_code'] == '0:0'
for name, proof in verified['registrations'].items():
    assert proof['status'] == 'PASS'
    cpu_path = root / f'docs/research/semantic_recovery_evidence/craftax_cpu/{name}-cpu-acceptance.json'
    cpu = json.loads(cpu_path.read_text())
    assert cpu['cpu_tests'] == proof['tests'] and cpu['exact_array_cases'] == 11
    for rel, sha in cpu['artifacts'].items():
        assert hashlib.sha256((root / rel).read_bytes()).hexdigest() == sha
    common = [cpu_path, root / cpu['test_xml']['path'], e / 'verified-gpu.json', e / 'remote-terminal.txt', e / 'slurm-6679082.log', e / 'DOWNLOAD-SHA256SUMS', e / 'download-verification.json', e / 'preservation-before-promotion.json']
    paths = common + sorted((e / 'remote/evidence' / name).iterdir())
    paths += [e / 'remote/evidence' / (name + suffix) for suffix in ['.xml', '-tests.log', '-arrays.log']]
    paths += [e / 'remote/evidence' / n for n in ['GPU-SHA256SUMS', 'bundle.json', 'SHA256SUMS', 'nvidia-smi.txt', 'freeze.txt']]
    acceptance = {
        'historical_id': f'swm/{name}-v1', 'status': 'ACCEPTED', 'accepted_classes': [name],
        'engine': 'Classic' if 'Classic' in name else 'Full', 'presentation': 'pixels' if 'Pixels' in name else 'symbolic',
        'implementation_commit': '2953735e5eabae96b796843a376abd7150781de4',
        'tested_source_commit': 'b6a8242f5ff5a9cae5e9e883a15611d779c2c99c',
        'cpu_checkpoint_commit': '28e23907f887c74edb56471f4011bbb819cdda5e',
        'gpu_evidence_commit': '59ba52300cc2e9c05da2eea0a58b64c57008036f',
        'oracle_commit': '176d54532b3c0a251a4f509b74f2fdacc143a63d',
        'source_archive_sha256': verified['source_archive_sha256'],
        'remote_archive_sha256': '8f26f97ca5509606a32ca88f65224836ee70dbe67f759a80a2c1d4add6e2e932',
        'job': 6679082, 'cpu_tests': cpu['cpu_tests'], 'gpu_tests': proof['tests'],
        'cpu_exact_array_cases': 11, 'gpu_exact_array_cases': 11,
        'gates': {k:'PASS' for k in ['source_ownership','cpu_semantics','gpu_semantics','exhaustive_numeric_state','atomic_restoration','rng_replay_and_isolation','reward_and_lifecycle','presentation_contract','independent_download_hash_and_array_verification']},
        'gpu_profile': proof['profile'], 'cpu_profile': cpu['profile'],
        'scope': {'default_execution':'DELEGATED_CPU','accepted_devices':['cpu','cuda'], 'gpu_engine':'delegated real upstream JAX','configuration':'fixed registered no-auto-reset defaults','public_execution':'JIT scalar upstream kernels per batch lane','tested_batches':[1,3],'seeds':[0,17,42], 'variation':'NONE', 'state_schema':cpu['schema'], 'gpu_schema':str((e/'remote/evidence'/name/'schema.json').relative_to(root)), 'cross_device_equality':'NOT_CLAIMED', 'unsupported':['MPS','public eager execution','vmap equivalence','arbitrary parameters','alternate rendering sizes','supersampling'] + (['pixel rendering'] if 'Symbolic' in name else [])},
        'promotion_is_independent': True, 'other_registration_evidence_required': False,
        'artifacts':[{'path':str(p.relative_to(root)), 'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in paths],
        'production_swm_imports':False,'copied_game_logic':False,'shared_contract_change':False,
    }
    target = e / (name + '-acceptance.json')
    assert not target.exists()
    target.write_text(json.dumps(acceptance,indent=2)+'\n')

for filename in ['semantic_recovery_matrix.json','semantic_recovery_implementation_map.json']:
    p = root/'docs/research'/filename
    d = json.loads(p.read_text())
    for row in d['rows']:
        if row['family'] != 'craftax':
            continue
        name = row['candidate_classes'][0]
        assert verified['registrations'][name]['status'] == 'PASS'
        manifest = str((e/(name+'-acceptance.json')).relative_to(root))
        if filename == 'semantic_recovery_matrix.json':
            row['state_status']='VERIFIED'
            row['adapter_status']='IMPLEMENTED_CPU_GPU_ACCEPTED'
            row['accepted_classes']=[name]
            row['accepted_devices']=['cpu','cuda']
            for key in ['action_parity','reset_parity','step_parity','reward_parity','terminated_truncated_parity','rng_lifecycle','renderer_regression']:
                row[key]=row[key].replace('PASS_EXACT_CPU','PASS_EXACT_CPU_GPU')
            row['stable_gpu_benchmark']=str((e/'remote/evidence'/name/'result.json').relative_to(root))
            row['remaining_issues']=['Fixed registered defaults only; no Variation, public eager/vmap, arbitrary parameters, MPS or alternate rendering sizes', 'CPU/GPU cross-device equality/restoration unclaimed; independent per-device exact oracle agreement accepted','Legacy frozen crosswalk/atlas classes preserved; accepted_classes identifies the recovered implementation']
            row['audit_report']='docs/research/craftax_acceptance.md'
            row['recovery_status']='ACCEPTED_INDEPENDENT_CPU_GPU_GATES'
            row['evidence'].append(manifest)
        else:
            row['current_status']='VERIFIED'
            row['accepted_classes']=[name]
            row['accepted_devices']=['cpu','cuda']
            row['upstream']['version_scope']='Pinned CPU and GPU profiles independently accepted; no cross-device equality claim'
            row['state_schema']['status']='CPU_GPU_PROVED'
            row['accepted_variation_scope']='NONE_REGISTERED_FIXED_DEFAULTS_CPU_GPU_VALIDATED'
            row['gpu_semantic_result']={'status':'PASS_EXACT_GPU','job':6679082,'terminal_state':'COMPLETED','exit_code':'0:0','node':'gpu2110','acceptance_manifest':manifest,'gpu_tests':verified['registrations'][name]['tests'],'exact_array_cases':11}
            row['frozen_renderer_result']=row['frozen_renderer_result'].replace('PASS_EXACT_CPU','PASS_EXACT_CPU_GPU')
            row['evidence_manifest']['artifacts'].append(manifest)
            row['known_limitations'][0]='CPU/GPU equality unclaimed; exact source agreement accepted independently per device'
            row['remaining_gates']=[]
    if filename == 'semantic_recovery_matrix.json':
        d['craftax_acceptance']={'report':'docs/research/craftax_acceptance.md','job':6679082,'gpu_evidence_commit':'59ba52300cc2e9c05da2eea0a58b64c57008036f','counts':{'VERIFIED':39,'INCOMPLETE':3},'independent_registration_gates':True}
    else:
        d['current_counts']={'VERIFIED':39,'INCOMPLETE':3}
        d['updates'].append({'checkpoint':'Craftax independent CPU/GPU acceptance','job':6679082,'report':'docs/research/craftax_acceptance.md','newly_verified':[f'swm/{n}-v1' for n in verified['registrations']],'counts':{'VERIFIED':39,'INCOMPLETE':3}})
    p.write_text(json.dumps(d,indent=2)+'\n')
p=root/'docs/research/index.md'; p.write_text(p.read_text()+'- [Craftax final CPU/GPU acceptance](craftax_acceptance.md)\n')
